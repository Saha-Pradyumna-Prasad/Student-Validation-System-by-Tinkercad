Student Validation System

Type a 3-digit student ID on the keypad and press #. An Arduino UNO checks it against its database, then answers with a message on the LCD, an LED and a buzzer tone.

How it works

The system validates student IDs. The Arduino compares the entered ID with a built-in list of students. A valid ID lights the green LED, plays a success tone and shows the student's details. An invalid ID flashes the red LED with an error tone.

  1. Enter the IDType 3 digits on the 4x4 keypad.
  2. SubmitPress # to send the ID.
  3. CheckThe Arduino searches its student database.
  4. RespondValid: green LED and success beep. Invalid: red LED and error beep.

Key features

Real-time validation

Verifies a 3-digit ID the moment you press #.

Sound and light feedback

Green LED and a rising tone for valid IDs. Red LED and a low alarm for invalid ones.

Built-in student records

Stores each student's name, department and payment status.

Clear LCD messages

Shows the student's details or a plain error message on the 16x2 display.

Circuit design

Arduino source code

StudentValidation.ino 
#include <LiquidCrystal.h>
#include <Keypad.h>

// LCD pin setup (RS, E, D4, D5, D6, D7)
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

// Keypad configuration (4x3 Keypad)
const byte ROWS = 4; 
const byte COLS = 3; 
char keys[ROWS][COLS] = {
  {'1','2','3'},
  {'4','5','6'},
  {'7','8','9'},
  {'*','0','#'}
};

byte rowPins[ROWS] = {9, 8, 7, 6}; 
byte colPins[COLS] = {A3, A2, A1}; 

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// Pin configurations
const int buzzerPin = 13;
const int greenLed = A4;
const int redLed = A5;

// Database structure
struct Student {
  String id;
  String name;
  String dept;
  String payment;
};

// You can add more students here
Student students[] = {
  {"112", "Rafi", "CSE", "Paid"},
  {"123", "Renu", "EEE", "Due"},
  {"526", "Alvi", "BBA", "Paid"},
  {"286", "Ria", "IT", "Due"}
};
int totalStudents = 4;

String enteredID = "";

void setup() {
  pinMode(buzzerPin, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(redLed, OUTPUT);
  
  lcd.begin(16, 2);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Smart ID System");
  lcd.setCursor(0, 1);
  lcd.print("Enter 3-Digit ID");
  delay(2000);
  showHomeScreen();
}

void loop() {
  char key = keypad.getKey();
  
  if (key) {
    // Clear input if '*' is pressed
    if (key == '*') {
      enteredID = "";
      showHomeScreen();
      return;
    }
    
    // Start search if '#' is pressed
    if (key == '#') {
      if (enteredID.length() == 3) {
        verifyStudent(enteredID);
      } else {
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Error: Need 3-ID");
        digitalWrite(redLed, HIGH);
        tone(buzzerPin, 1000, 500);
        delay(1500);
        digitalWrite(redLed, LOW);
        showHomeScreen();
      }
      enteredID = "";
      return;
    }
    
    // While typing numbers
    if (isDigit(key)) {
      if (enteredID.length() < 3) {
        enteredID += key;
        lcd.setCursor(0, 1);
        lcd.print("ID: " + enteredID);
      }
    }
  }
}

void showHomeScreen() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Scan ID & Press#");
  lcd.setCursor(0, 1);
  lcd.print("ID: ");
}

void verifyStudent(String id) {
  bool found = false;
  lcd.clear();
  
  for (int i = 0; i < totalStudents; i++) {
    if (students[i].id == id) {
      found = true;
      // If successful, green LED and positive beep sound
      digitalWrite(greenLed, HIGH);
      tone(buzzerPin, 2000, 200);
      delay(300);
      tone(buzzerPin, 2500, 200);
      
      // Display student information and Valid status
      lcd.setCursor(0, 0);
      lcd.print(students[i].name + " (" + students[i].dept + ")");
      lcd.setCursor(0, 1);
      lcd.print("Status: Valid");
      
      delay(4000); // Keep info on display for 4 seconds
      digitalWrite(greenLed, LOW);
      break;
    }
  }
  
  if (!found) {
    // If ID doesn't match, red LED and error beep sound
    digitalWrite(redLed, HIGH);
    tone(buzzerPin, 500, 800); // Low pitch beep sound
    
    lcd.setCursor(0, 0);
    lcd.print("ID: " + id);
    lcd.setCursor(0, 1);
    lcd.print("Invalid ID!");
    
    delay(3000);
    digitalWrite(redLed, LOW);
  }
  
  showHomeScreen();
}

Challenges and learnings

  • Handling keypad debounce and correct pin mapping.
  • Updating the LCD without flicker.
  • Using a C++ struct to keep the student database tidy.

Ideas for the next version

  • Keep records in EEPROM or on an SD card instead of hard-coding them.
  • Read IDs from an RFID card instead of typing them.
  • Log every scan with a time stamp using an RTC module.
  • Lock the keypad for a short time after 3 wrong attempts.
