How it Works

This system is designed to validate student IDs. A user enters a 3-digit ID using the 4x4 Keypad and presses '#' to submit. The Arduino cross-references the ID with a pre-defined database. If the ID is valid, the Green LED lights up, a success tone plays, and student details are shown on the LCD. If invalid, the Red LED flashes with an error tone.

Key Features

  • Real-time Validation: Instantly verifies 3-digit IDs.
  • Audio-Visual Feedback: Green LED + Success Tone for valid, Red LED + Alarm for invalid.
  • Database Integration: Stores Student Name, Dept, and Payment Status.
  • LCD Display: Shows real-time data and error messages.

Working Principle

User inputs a 3-digit ID via Keypad.
Presses '#' to submit the ID.
Arduino checks the internal database.
Valid ID -> Green LED & Success Beep.
Invalid ID -> Red LED & Error Beep.

Arduino Source Code

StudentValidation.ino 

#include <LiquidCrystal.h>
#include <Keypad.h>

// LCD pin setup (RS, E, D4, D5, D6, D7)
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

// Keypad configuration (4x3 Keypad)
const byte ROWS = 4; 
const byte COLS = 3; 
char keys[ROWS][COLS] = {
  {'1','2','3'},
  {'4','5','6'},
  {'7','8','9'},
  {'*','0','#'}
};

byte rowPins[ROWS] = {9, 8, 7, 6}; 
byte colPins[COLS] = {A3, A2, A1}; 

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// Pin configurations
const int buzzerPin = 13;
const int greenLed = A4;
const int redLed = A5;

// Database structure
struct Student {
  String id;
  String name;
  String dept;
  String payment;
};

// You can add more students here
Student students[] = {
  {"159", "Ankur", "CSE", "Paid"},
  {"113", "Manik", "EEE", "Due"},
  {"153", "Tina", "BBA", "Paid"},
  {"120", "Mahi", "IT", "Due"}
};
int totalStudents = 4;

String enteredID = "";

void setup() {
  pinMode(buzzerPin, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(redLed, OUTPUT);
  
  lcd.begin(16, 2);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Smart ID System");
  lcd.setCursor(0, 1);
  lcd.print("Enter 3-Digit ID");
  delay(2000);
  showHomeScreen();
}

void loop() {
  char key = keypad.getKey();
  
  if (key) {
    // Clear input if '*' is pressed
    if (key == '*') {
      enteredID = "";
      showHomeScreen();
      return;
    }
    
    // Start search if '#' is pressed
    if (key == '#') {
      if (enteredID.length() == 3) {
        verifyStudent(enteredID);
      } else {
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Error: Need 3-ID");
        digitalWrite(redLed, HIGH);
        tone(buzzerPin, 1000, 500);
        delay(1500);
        digitalWrite(redLed, LOW);
        showHomeScreen();
      }
      enteredID = "";
      return;
    }
    
    // While typing numbers
    if (isDigit(key)) {
      if (enteredID.length() < 3) {
        enteredID += key;
        lcd.setCursor(0, 1);
        lcd.print("ID: " + enteredID);
      }
    }
  }
}

void showHomeScreen() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Scan ID & Press#");
  lcd.setCursor(0, 1);
  lcd.print("ID: ");
}

void verifyStudent(String id) {
  bool found = false;
  lcd.clear();
  
  for (int i = 0; i < totalStudents; i++) {
    if (students[i].id == id) {
      found = true;
      // If successful, green LED and positive beep sound
      digitalWrite(greenLed, HIGH);
      tone(buzzerPin, 2000, 200);
      delay(300);
      tone(buzzerPin, 2500, 200);
      
      // Display student information and Valid status
      lcd.setCursor(0, 0);
      lcd.print(students[i].name + " (" + students[i].dept + ")");
      lcd.setCursor(0, 1);
      lcd.print("Status: Valid");
      
      delay(4000); // Keep info on display for 4 seconds
      digitalWrite(greenLed, LOW);
      break;
    }
  }
  
  if (!found) {
    // If ID doesn't match, red LED and error beep sound
    digitalWrite(redLed, HIGH);
    tone(buzzerPin, 500, 800); // Low pitch beep sound
    
    lcd.setCursor(0, 0);
    lcd.print("ID: " + id);
    lcd.setCursor(0, 1);
    lcd.print("Invalid ID!");
    
    delay(3000);
    digitalWrite(redLed, LOW);
  }
  
  showHomeScreen();
}
